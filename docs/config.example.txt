docker run -d --name odoo-mcp-server `
  -p 8000:8000 `
  -e ODOO_URL=http://host.docker.internal:8069 `
  -e ODOO_DB=your-database `
  -e ODOO_USER=your-username `
  -e ODOO_PASSWORD=your-password `
  -e ODOO_YOLO=true `
  odoo-mcp --transport sse --host 0.0.0.0 --port 8000
